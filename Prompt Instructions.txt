public float Convert_Half_Life_To_Cumulation_Weight(float processing_window_duration, float half_life)
{
    if (Scenario_Information.Should_Validate && (half_life_duration < 0.00000001f))
    {
        Logger?.Error($"Attempting to calculate Convert_Half_Life_To_Cumulation_Weight() when half_life {half_life} is meaningfully close to or below 0.0");
    }

    float processing_window_to_halflife_duration = (processing_window_duration / half_life);

    float cumulation_weight = (float)Math.Pow(0.5f, processing_window_to_halflife_duration);

    if ((cumulation_weight < 0.0f) || (cumulation_weight > 1.0f))
    {
        Logger?.Error($"Convert_Half_Life_To_Cumulation_Weight() calculated a cumulation_weight {cumulation_weight} outside of 0.0->1.0 bounds");
    }
    return (cumulation_weight);
}
